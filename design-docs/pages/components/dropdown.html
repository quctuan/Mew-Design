Dropdown

Hiển thị danh sách các hành động hoặc tùy chọn khi nhấp vào một thành phần, được sử dụng cho menu điều hướng hoặc lệnh.

Preview

My Account

Usage

iOS (Swift) 
import MyVNPTDesignSystem

// Dropdown (Menu / ActionSheet)
DSDropdown(
    title: "Options",
    items: [
        DSDropdownItem(title: "Profile", icon: nil),
        DSDropdownItem(title: "Billing", icon: nil),
        DSDropdownItem(title: "Delete", isDestructive: true)
    ],
    onSelect: { item in
        print("Selected \(item.title)")
    }
)
Android (Kotlin) 
import vn.myvnpt.designsystem.components.DSDropdown
import vn.myvnpt.designsystem.components.DSDropdownItem

// Dropdown Menu
DSDropdown(
    title = "Options",
    items = listOf(
        DSDropdownItem(title = "Profile"),
        DSDropdownItem(title = "Billing"),
        DSDropdownItem(
            title = "Delete",
            isDestructive = true
        )
    ),
    onItemClick = { item ->
        // Handle selection
    }
)
